cmake_minimum_required(VERSION 3.28)
project(classdump CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(SLANG_INCLUDE_TOOLS OFF CACHE BOOL "" FORCE)
set(SLANG_INCLUDE_TESTS OFF CACHE BOOL "" FORCE)
set(SLANG_INCLUDE_PYLIB OFF CACHE BOOL "" FORCE)
set(SLANG_INCLUDE_DOCS  OFF CACHE BOOL "" FORCE)
# SLANG_DIR is passed in by build_slang.sh (a checkout of slang at the pinned commit)
add_subdirectory(${SLANG_DIR} slang EXCLUDE_FROM_ALL)
add_executable(classdump classdump.cpp)
target_link_libraries(classdump PRIVATE slang::slang)
